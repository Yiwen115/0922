function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(15000); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(["日期時間", "校園地點", "GPS", "物種", "數量與單位", "照片縮圖", "照片原始連結", "備註"]); sheet.getRange("A1:H1").setFontWeight("bold").setBackground("#e2e8f0"); sheet.setColumnWidth(6, 120); } var data = JSON.parse(e.postData.contents); var datetime = data.datetime || Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy-MM-dd HH:mm:ss"); var location = data.location || ""; var gps = data.gps || ""; var species = data.species || ""; var quantity = data.quantity || ""; var notes = data.notes || ""; var photoViewUrl = ""; var imageFormula = ""; // 儲存照片到 Google Drive if (data.fileBase64 && data.fileName) { var folderName = "校園花朵照片"; var folders = DriveApp.getFoldersByName(folderName); var targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); var decodedBytes = Utilities.base64Decode(data.fileBase64); var blob = Utilities.newBlob(decodedBytes, data.fileMime || "image/jpeg", data.fileName); var file = targetFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoViewUrl = file.getUrl(); var directDownloadUrl = "https://lh3.googleusercontent.com/d/" + file.getId(); imageFormula = '=IMAGE("' + directDownloadUrl + '")'; } sheet.appendRow([datetime, location, gps, species, quantity, imageFormula, photoViewUrl, notes]); if (imageFormula) { sheet.setRowHeight(sheet.getLastRow(), 80); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { if (lock.hasLock()) lock.releaseLock(); } } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: "online" })).setMimeType(ContentService.MimeType.JSON); }
